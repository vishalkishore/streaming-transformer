cmake_minimum_required(VERSION 3.24)
set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "GPU architectures (120: Blackwell target; 86: previous Ampere target)")
project(Stream LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(GTest REQUIRED)
add_library(stream src/runtime/Model.cpp src/runtime/Provenance.cpp src/cpu/Cpu.cpp src/cuda/Runtime.cu)
target_include_directories(stream PUBLIC include PRIVATE src/cuda)
string(REPLACE ";" "," STREAM_CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
target_compile_definitions(stream PRIVATE STREAM_CUDA_ARCHITECTURES="${STREAM_CUDA_ARCHITECTURES}")
target_link_libraries(stream PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json CUDA::cudart CUDA::cublas OpenSSL::Crypto Threads::Threads)
target_compile_options(stream PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-march=native;-Wall;-Wextra>
  PUBLIC $<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr;-lineinfo;--ptxas-options=-v;--fmad=false>)
set_target_properties(stream PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
add_executable(stream-bench bench/Main.cpp)
target_link_libraries(stream-bench PRIVATE stream)
add_executable(stream-probe bench/Probe.cu)
target_link_libraries(stream-probe PRIVATE stream)
add_executable(stream-diagnose bench/Diagnose.cpp)
target_link_libraries(stream-diagnose PRIVATE stream)
enable_testing()
add_executable(stream-tests tests/Native.cpp tests/Primitives.cu tests/Lifecycle.cu)
target_include_directories(stream-tests PRIVATE src/cuda)
target_link_libraries(stream-tests PRIVATE stream GTest::gtest_main)
add_test(NAME native COMMAND stream-tests)
set_tests_properties(native PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(stream-cpu-noalloc-tests tests/CpuAllocation.cpp src/cpu/Cpu.cpp src/runtime/Model.cpp)
target_include_directories(stream-cpu-noalloc-tests PRIVATE include)
target_link_libraries(stream-cpu-noalloc-tests PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json OpenSSL::Crypto GTest::gtest_main)
target_compile_definitions(stream-cpu-noalloc-tests PRIVATE EIGEN_RUNTIME_NO_MALLOC)
target_compile_options(stream-cpu-noalloc-tests PRIVATE -O3 -march=native -UNDEBUG)
add_test(NAME cpu-noalloc COMMAND stream-cpu-noalloc-tests)
set_tests_properties(cpu-noalloc PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
